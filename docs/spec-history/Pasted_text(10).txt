                         for rid, t in self._tasks.items()
                         if t["state"] is TaskState.RUNNING]
            queued = [rid for rid, t in self._tasks.items()
                      if t["state"] is TaskState.QUEUED]
            for rid in queued:
                self._tasks[rid]["state"] = TaskState.CANCELED_AT_RECOVERY
            return in_flight, queued

    def _worker(self) -> None:
        while True:
            item = self._queue.get()
            try:                                   # B3: task_done auf JEDEM Pfad
                if item is self.SENTINEL:
                    return
                rid, op, fn = item
                with self._lock:
                    task = self._tasks.get(rid)
                    if task is None:
                        canceled = False           # bereits finalisiert
                    else:
                        canceled = task["state"] is TaskState.CANCELED_AT_RECOVERY
                        if not canceled:
                            task["state"] = TaskState.RUNNING
                # B1: Lock FREIGEBEN vor _finalize
                if canceled:
                    if self._finalize(rid) and self._on_canceled:
                        try:
                            self._on_canceled(rid, op)
                        except BaseException as e:
                            if self._on_worker_crash:
                                self._on_worker_crash(rid, op, type(e).__name__)
                    continue
                try:
                    fn(rid, op)
                except BaseException as e:         # Worker-Thread stirbt nie
                    if self._on_worker_crash:
                        self._on_worker_crash(rid, op, type(e).__name__)
            finally:
                self._queue.task_done()

    def shutdown(self, cancel_queued: bool = True,

@dataclass
class CleanupReport:
    state: str
    resources: list = field(default_factory=list)
    console_dirty: bool = False


class ResourceManager:
    def __init__(self) -> None:
        self._lock = threading.RLock()
        self._held: dict[str, ManagedResource] = {}
        self._leaked: dict[str, ManagedResource] = {}

    def acquire(self, kind: str, *, recoverable: bool = True,
                force_cleanup_failure: bool = False,
                out_of_band_safe: bool = False) -> str:
        with self._lock:
            ref = f"{kind}:{os.urandom(8).hex().upper()}"
            self._held[ref] = ManagedResource(
                ref=ref, kind=kind, recoverable=recoverable,
                out_of_band_cleanup_safe=out_of_band_safe,
                force_cleanup_failure=force_cleanup_failure)
            return ref

    def attach_cleanup(self, ref: str, callback: Callable[[], bool]) -> None:
        """B12: Callback NACH acquire an den konkreten Ref binden.
        Kein Callback mit leerem/fremdem Ref; Ref muss existieren."""
        with self._lock:
            res = self._held.get(ref)
            if res is None:
                raise ResourceStateInvalid(f"attach_cleanup auf unbekannt {ref!r}")
            res.cleanup = callback

    def release(self, ref: str) -> bool:
        with self._lock:
            r = self._held.get(ref)
            if r is None:
                return False
            result = r.do_cleanup()
            r.cleanup_result = result
            if result is CleanupResult.FREED:
                del self._held[ref]
                return True
            self._leaked[ref] = self._held.pop(ref)
            return False

    def cleanup_all(self) -> CleanupReport:
        """Normaler Disconnect: alle gehaltenen Ressourcen."""
        with self._lock:
            for ref in list(self._held):
                self.release(ref)
            return self._report()

    def cleanup_after_hang(self) -> CleanupReport:
        """V5/F10: nur out_of_band_cleanup_safe bereinigen. Alles andere:
        cleanup_attempted=False (KEIN Versuch), UNCERTAIN, console_dirty."""
        with self._lock:
            for ref, r in list(self._held.items()):
                if r.out_of_band_cleanup_safe:
                    result = r.do_cleanup()
                    r.cleanup_result = result
                    if result is CleanupResult.FREED:
                        del self._held[ref]
                    else:
                        self._leaked[ref] = self._held.pop(ref)
                else:
                    r.cleanup_attempted = False
                    r.cleanup_result = CleanupResult.UNCERTAIN
                    self._leaked[ref] = self._held.pop(ref)
            return self._report()

    def _report(self) -> CleanupReport:
        resources = [{"ref": r.ref, "kind": r.kind,
                      "cleanup_attempted": r.cleanup_attempted,
                      "cleanup_result": (r.cleanup_result.value
                                         if r.cleanup_result else "NOT_ATTEMPTED"),
                      "recoverable": r.recoverable,
                      "out_of_band_safe": r.out_of_band_cleanup_safe}
                     for r in self._leaked.values()]
        dirty = bool(self._leaked)
        return CleanupReport(state="COMPLETE" if not dirty else "PARTIAL",
                             resources=resources, console_dirty=dirty)
