@property
def available_tokens(self) -> int:
    """Read-only Bestand via öffentlicher API: zählt reservierbare Tokens

    for _ in range(n):

    return n

§7 Gate-Runner — Punkte 7–9 mechanisch erzwungen

tools/freeze/run_gate_suite.py — führt die Suite real aus, schreibt den Report, taggt nur bei Grün und sauberem Worktree:
python
 
  
 
 
#!/usr/bin/env python3
"""PS4RF Freeze Gate Runner. Tag nur wenn ALLE Gates grün UND Worktree sauber.
Bei Fehlern: FAILED + Testname + Exception (Punkt 8) — kein Tag."""
import json, subprocess, sys
from datetime import datetime, timezone
from pathlib import Path

TAG = "PS4RF-CORE-1.0-MOCK-READY"
GATES = [
    ("G1", ["python", "-m", "pytest", "tests/", "-q", "--tb=short"]),
    ("G2", ["python", "-m", "pytest", "tests/test_roundtrip.py", "-q"]),
    ("G3", ["python", "-m", "pytest", "tests/test_mock_recovery.py", "-q"]),
    ("G4", ["python", "-m", "pytest", "tests/test_stress.py",
            "tests/test_concurrency.py", "tests/test_pool.py", "-q"]),
    ("G5", ["python", "-m", "pytest", "tests/test_gate.py", "tests/test_sync.py", "-q"]),
    ("G6", ["python", "-m", "pytest", "tests/test_verification.py", "-q"]),
    ("G7", ["python", "-m", "pytest", "tests/test_framing.py",
            "tests/test_handshake_roles.py", "-q"]),
    ("G8", ["python", "-m", "pytest", "tests/test_rpc_budget.py", "-q"]),
]

def sh(cmd):
    r = subprocess.run(cmd, capture_output=True, text=True)
    return r.returncode, (r.stdout + r.stderr).strip()

def main():
    results, failures = {}, []
    for gate, cmd in GATES:
        rc, out = sh(cmd)
        summary = out.splitlines()[-1] if out else ""
        results[gate] = {"cmd": " ".join(cmd), "rc": rc, "summary": summary}
        if rc != 0:
            failures.append((gate, out))
    commit = sh(["git", "rev-parse", "HEAD"])[1]
    dirty = sh(["git", "status", "--porcelain"])[1]
    report = {"tag_candidate": TAG, "commit": commit, "worktree_dirty": bool(dirty),
              "python": sys.version, "timestamp_utc": datetime.now(timezone.utc).isoformat(),
              "gates": results, "all_green": not failures and not dirty}
    Path("FREEZE-REPORT.json").write_text(json.dumps(report, indent=2))
    print(json.dumps(report, indent=2))
    for gate, out in failures:                      # Punkt 8: FAILED-Report
        print(f"\nFAILED gate={gate}\n{out[-4000:]}")
    if report["all_green"]:
        subprocess.run(["git", "tag", TAG], check=True)   # Punkt 4: korrekter Tag
        print(f"\nTAG gesetzt: {TAG} @ {commit}")
        return 0
    print("\nNICHT getaggt: Gate-Fehler oder dirty worktree.")
    return 1

if __name__ == "__main__":
    sys.exit(main())

§8 Was ich hier nicht liefern kann — und warum

Einen ausgefüllten Report mit Commit-Hash, Testzahlen und „G1–G8 PASS" kann ich nicht ehrlich erzeugen: Es existiert hier keine Ausführungsumgebung. Fingierte Werte wären genau die Hash-ohne-Run-Behauptung, die dieses Framework ausschließt. Die Suite muss einmal real laufen (Kommandos in §7); bei Fehlern liefern mir die FAILED-Blöcke des Runners Testname + Exception + Traceback, und ich liefere Root Cause + minimalen Fix — ohne neue Architektur, wie in Punkt 8 gefordert.
