def merge_boot_session(name: str, context: TargetExecutionContext,
                       boot_id: str, session_id: str,
                       boot_res: "Resolution", sess_res: "Resolution") -> "Resolution":
    """F4 — normativ:
    Availability: SESSION-Befund in {AVAILABLE, DEGRADED, UNAVAILABLE} gewinnt.
    SESSION UNKNOWN oder NO_RECORD → BOOT-Availability (fehlendes Session-Wissen
    vernichtet keinen belegten Boot-Zustand nicht).
    Assurance: max(BOOT, SESSION) — nur gültig, da resolve() (R1) bereits
    exakte Identität (name, context, boot_id) erzwingt; Guard unten.
    Provenance: die Resolution der gewählten Quelle wird übernommen."""
    for res, sid in ((boot_res, None), (sess_res, session_id)):
        if res.name != name:
            raise ValueError("Merge-Identität verletzt: name")
        # context/boot_id sind in resolve() bereits strukturell gefiltert.
    sess_has = sess_res.availability_resolution.factor is not AxisFactor.NO_RECORD
    if sess_has and sess_res.availability in _SESSION_EFFECTIVE:
        avail, a_prov = sess_res.availability, sess_res.availability_resolution
    else:
        avail, a_prov = boot_res.availability, boot_res.availability_resolution
    if ASSURANCE_RANK[boot_res.assurance] >= ASSURANCE_RANK[sess_res.assurance]:
        assur, s_prov = boot_res.assurance, boot_res.assurance_resolution
    else:
        assur, s_prov = sess_res.assurance, sess_res.assurance_resolution
    return Resolution(name, avail, assur, a_prov, s_prov)

    def poison(self, reason: str) -> tuple[list[dict], list[str]]:
        """Rückgabe: (in_flight, queued_rids). Queued werden als
        CANCELED_AT_RECOVERY markiert; Worker erkennen beim Dequeue."""
        with self._lock:
            self._poisoned = True
            in_flight = [{"request_id": rid, "operation": t["op"],
                          "status": "IN_FLIGHT_AT_RECOVERY"}
                         for rid, t in self._tasks.items()
                         if t["state"] is TaskState.RUNNING]
            queued = [rid for rid, t in self._tasks.items()
                      if t["state"] is TaskState.QUEUED]
            for rid in queued:
                self._tasks[rid]["state"] = TaskState.CANCELED_AT_RECOVERY
            return in_flight, queued

    # ---------- Worker ----------
    def _worker(self) -> None:
        while True:
            item = self._queue.get()
            if item is self.SENTINEL:                     # F9
                return
            rid, op, fn = item
            with self._lock:
                task = self._tasks.get(rid)
                if task is None:
                    continue                              # bereits finalisiert
                if task["state"] is TaskState.CANCELED_AT_RECOVERY:   # F7
                    if self._finalize(rid, TaskState.CANCELED_AT_RECOVERY) \
                            and self._on_canceled:
                        self._on_canceled(rid, op)        # F7: Respond falls Transport ok
                    continue
                task["state"] = TaskState.RUNNING
            fn(rid, op)                                    # F11: fn ruft _finish im finally

    # garantiert nur Token/State. Agent-_worker_run:
    #   try handler / except expected / except unexpected → INTERNAL_ERROR
    #   (falls nicht poisoned & Transport ok, sonst Evidence only)
    #   finally: _finalize(rid, COMPLETED) + registry.complete(rid)

    def finish(self, rid: str) -> None:                    # nach erfolgreichem Respond
        self._finalize(rid, TaskState.COMPLETED)

    def has_pending(self) -> bool:
        with self._lock:

    # ---------- F9: Shutdown ----------
    def shutdown(self, cancel_queued: bool = True, wait_running: bool = False,
                 cancel_cb: Optional[Callable[[str, str], None]] = None) -> None:
        """Normal: queued fertig ODER canceln gemäß Close-Policy.
        Recovery/Hang: NIE auf gehangene Worker warten (wait_running=False)."""
        if cancel_queued:

                try:
                    item = self._queue.get_nowait()

                rid, op, _fn = item

                    cancel_cb(rid, op)

            self._queue.put(self.SENTINEL)

                t.join(timeout=5.0)
        # wait_running=False: Worker enden nach Sentinels bzw. nach

class Gate:
    def __init__(self, expected: int) -> None:
        self._cond = threading.Condition()
        self._participants: set[str] = set()          # F12
        self._expected = expected
        self._released = False
        self._destroyed = False

    def arrive(self, participant_id: str) -> int:
        with self._cond:
            if self._destroyed: raise GateDestroyed()
            if participant_id in self._participants:  # F12: Duplicate-Lane
                raise ResourceStateInvalid("duplicate participant arrival")
            self._participants.add(participant_id)
            if len(self._participants) >= self._expected and not self._released:
                self._released = True                 # expected basiert auf UNIQUE participants
                self._cond.notify_all()
            self._cond.notify_all()
            return len(self._participants)

    def status(self) -> dict:
        with self._cond:
            return {"arrived_count": len(self._participants),
                    "expected": self._expected,
                    "released": self._released, "destroyed": self._destroyed,
                    # participant IDs nur in Debug/Evidence:
                    "participants": sorted(self._participants)}
