@dataclass(frozen=True)
class ManualOverride:
    availability: Optional[Availability] = None
    assurance: Optional[Assurance] = None
    evidence: Optional[EvidenceRef] = None
    verified_by: Optional[VerifiedBy] = None
    detail: str = ""

class AxisFactor(str, Enum):
    MANUAL_OVERRIDE = "MANUAL_OVERRIDE"; PROVIDER_AUTHORITY = "PROVIDER_AUTHORITY"
    RUNTIME_RECENCY = "RUNTIME_RECENCY"; RECENCY = "RECENCY"
    EVIDENCE_RANK = "EVIDENCE_RANK"; AUTHORITY_CEILING = "AUTHORITY_CEILING"
    NO_RECORD = "NO_RECORD"

@dataclass(frozen=True)
class AxisResolution:
    value: str; factor: AxisFactor
    record_index: Optional[int] = None; detail: str = ""

@dataclass(frozen=True)
class Resolution:
    name: str; availability: Availability; assurance: Assurance
    availability_resolution: AxisResolution; assurance_resolution: AxisResolution

def _clamp(a: Assurance, ceiling: Assurance) -> Assurance:
    return a if ASSURANCE_RANK[a] <= ASSURANCE_RANK[ceiling] else ceiling

def resolve(name: str, records: Sequence[CapabilityRecord],
            authority: AuthorityTable,
            manual: Optional[ManualOverride] = None) -> Resolution:
    indexed = []
    for i, rec in enumerate(records):
        if rec.observed_seq is None:
            raise ValueError(f"Record {i} nicht durch CapabilityIntake ingestiert")
        rule = authority.rule_for(name, rec.provider_type, rec.provider)
        if rule is not None:
            indexed.append((i, rec, rule))

    # ---------- Availability: MANUAL → Authority → neuester Befund (observed_seq) ----------
    if manual is not None and manual.availability is not None:
        availability = manual.availability
        a_res = AxisResolution(availability.value, AxisFactor.MANUAL_OVERRIDE,
                               detail=manual.detail)
    else:
        elig = [(i, r, ru) for i, r, ru in indexed if ru.availability_write]
        if not elig:
            availability, a_res = Availability.UNKNOWN, AxisResolution(
                "UNKNOWN", AxisFactor.NO_RECORD)
        else:
            top = max(ru.availability_rank for _, _, ru in elig)
            # C1: Recency ausschließlich über observed_seq; provider_created_at
            # fließt nie ein (Manipulationssicherheit).
            tier = sorted([(i, r, ru) for i, r, ru in elig
                           if ru.availability_rank == top],
                          key=lambda t: t[1].observed_seq)
            i, r, _ = tier[-1]
            factor = (AxisFactor.PROVIDER_AUTHORITY if len(tier) == 1
                      else AxisFactor.RUNTIME_RECENCY)
            availability = r.availability
            a_res = AxisResolution(availability.value, factor, i)

    # ---------- Assurance: Manual-Grenze → Ceiling → EVIDENCE_RANK → RECENCY(seq) ----------
    best_rank, best = -1, None
    for i, r, ru in indexed:
        eff = _clamp(r.assurance, ru.max_assurance)
        if ASSURANCE_RANK[eff] > best_rank:
            best_rank, best = ASSURANCE_RANK[eff], (i, r, eff)
    manual_assur = None
    if manual is not None and manual.assurance is not None:
        if manual.assurance is Assurance.PROVEN and (
                manual.evidence is None or manual.verified_by is None):
            raise ValueError("MANUAL darf PROVEN nur mit evidence + verified_by setzen")
        op_rule = authority.rule_for(name, ProviderType.OPERATOR, "manual")
        manual_assur = _clamp(manual.assurance,
                              op_rule.max_assurance if op_rule else Assurance.OBSERVED)
    if manual_assur is not None and ASSURANCE_RANK[manual_assur] >= best_rank:
        assurance = manual_assur
        s_res = AxisResolution(assurance.value, AxisFactor.MANUAL_OVERRIDE)
    elif best is None:
        assurance, s_res = Assurance.UNVERIFIED, AxisResolution(
            "UNVERIFIED", AxisFactor.NO_RECORD)
    else:
        same = sorted([(i, r, _clamp(r.assurance, ru.max_assurance))
                       for i, r, ru in indexed
                       if ASSURANCE_RANK[_clamp(r.assurance, ru.max_assurance)] == best_rank],
                      key=lambda t: t[1].observed_seq)     # C1: recency via seq
        i, r, eff = same[-1]
        if len(same) > 1:                       factor = AxisFactor.RECENCY
        elif eff is not r.assurance:            factor = AxisFactor.AUTHORITY_CEILING
        else:                                   factor = AxisFactor.EVIDENCE_RANK
        assurance = eff
        s_res = AxisResolution(assurance.value, factor, i)

    return Resolution(name, availability, assurance, a_res, s_res)

    def _nonce(self, seq: int) -> bytes:
        return self._prefix + seq.to_bytes(8, "big")

    def _aad(self, seq: int, message_type: str) -> bytes:
        return build_aad(self._pv, self._sid, seq, self._dir, message_type)

    def seal(self, message_type: str, plaintext: bytes) -> tuple[int, bytes]:
        if self._send > LAST_SEQ:               # C2: 0..2^64-2 sendbar
            raise SequenceExhausted()
        seq = self._send
        self._send += 1
        return seq, self._aead.encrypt(self._nonce(seq), plaintext,
                                       self._aad(seq, message_type))

    def open(self, message_type: str, seq: int, ciphertext: bytes) -> bytes:
        if self._recv > LAST_SEQ:
            raise SequenceExhausted()           # Session tot → close + re-handshake
        if not isinstance(seq, int) or seq != self._recv:
            raise ReplayDetected(f"seq {seq!r}, erwartet {self._recv}")
        try:
            pt = self._aead.decrypt(self._nonce(seq), ciphertext,
                                    self._aad(seq, message_type))
        except InvalidTag as e:
            raise ProtocolError("AEAD verify failed") from e
        self._recv += 1
        return pt
 
 

Exhaustion-Semantik (vereinheitlicht, C2): gültige Sequence 0..2⁶⁴-2. seal() bei _send > LAST_SEQ → SequenceExhausted; open() bei _recv > LAST_SEQ → SequenceExhausted — Empfänger prüft Exhaustion vor der Sequence-Prüfung, da in diesem Zustand jede weitere Nachricht (Replay oder Peer-Verstoß) ohnehin Session-Abort bedeutet. Beide Richtungen identisch; Folge: Session schließen, neuer Handshake.

Wire-Format (AEAD-Phase, normativ): len(u32 BE) || msg_type(u8) || seq(u64 BE) || ciphertext. Nonce wird nicht übertragen. Falls eine künftige Variante ein Nonce-Feld trägt: received_nonce == expected_nonce, sonst PROTOCOL_ERROR + Session abort (vorgesehene Regel, aktuell ungenutzt).

def peak_tasks(steps) -> int:
    """C3: Peak = gleichzeitige Tasks.
    PARALLEL: sum(peak(branch_i)). Sequentiell: max(step peaks).
    REPEAT/IF: keine zusätzliche Parallelität (nur ausgeführter Body)."""
    peak = 1
    for s in steps:
        op = s.get("op")
        if op == "PARALLEL":
            peak = max(peak, sum(peak_tasks(_steps(b)) for b in _branches(s)))
        elif op == "IF":
            peak = max(peak, peak_tasks(_steps(s.get("then"))),
                       peak_tasks(_steps(s.get("else"))))
        elif op == "REPEAT_BOUNDED":
            peak = max(peak, peak_tasks(_steps(s.get("body"))))
    return peak

def nesting_depth(steps, d: int = 1) -> int:
    best = d
    for s in steps:
        for b in _subflows(s):
            best = max(best, nesting_depth(b, d + 1))
    return best

def runtime_bound_ms(steps, default_step_ms: int) -> int:
    t = 0
    for s in steps:
        op = s.get("op")
        if op == "REPEAT_BOUNDED":  t += s["count"] * runtime_bound_ms(_steps(s["body"]), default_step_ms)
        elif op == "IF":            t += max(runtime_bound_ms(_steps(s.get("then")), default_step_ms),
                                             runtime_bound_ms(_steps(s.get("else")), default_step_ms))
        elif op == "PARALLEL":      t += max((runtime_bound_ms(b, default_step_ms)
                                              for b in _branches(s)), default=0)
        elif op == "SLEEP_BOUNDED": t += s["ms"]
        elif op == "WAIT_BARRIER":  t += s.get("timeout_ms", default_step_ms)
        else:                       t += default_step_ms
    return t

def snapshot_bytes(steps, region_sizes=None) -> int:
    """C3: identische Kontrollfluss-Semantik wie expanded_steps."""
    t = 0
    for s in steps:
        op = s.get("op")
        if op == "REPEAT_BOUNDED":
            t += s["count"] * snapshot_bytes(_steps(s["body"]), region_sizes)
        elif op == "IF":
            t += max(snapshot_bytes(_steps(s.get("then")), region_sizes),
                     snapshot_bytes(_steps(s.get("else")), region_sizes))
        elif op == "PARALLEL":
            t += sum(snapshot_bytes(_steps(b), region_sizes) for b in _branches(s))
        elif op == "SNAPSHOT":
            t += sum(_range_size(r, region_sizes) for r in s.get("targets", []))
    return t

def check_budget(ir: dict, region_sizes: Optional[dict] = None) -> dict:
    computed = {
        "max_expanded_steps": expanded_steps(ir["steps"]),
        "max_parallel_tasks": peak_tasks(ir["steps"]),
        "max_nesting_depth": nesting_depth(ir["steps"]),
        "max_runtime_budget_ms": runtime_bound_ms(ir["steps"],
                                                  ir["timeouts"]["default_step_ms"]),
        "max_snapshot_bytes": snapshot_bytes(ir["steps"], region_sizes),
    }
    declared = ir.get("budget", {})
    # C3: max_rpc_bytes = declared hard cap (kein statisches Operation-Size-Modell);
    # Laufzeit-Enforcement zählt kumulative Response-Bytes.
    if "max_rpc_bytes" in declared and declared["max_rpc_bytes"] > CAPS["max_rpc_bytes"]:
        raise BudgetExceeded("max_rpc_bytes über globalem Cap")
    for k, cap in CAPS.items():
        if k == "max_rpc_bytes":
            continue
        v = computed[k]
        if v > cap or (k in declared and v > declared[k]):
            raise BudgetExceeded(f"{k}: computed={v}, cap={cap}, declared={declared.get(k)}")
    return computed

"""Operation Registry: Name → Capability-Pflicht. Quelle für Grant-Checks."""
from __future__ import annotations
from dataclasses import dataclass

@dataclass(frozen=True)
class OperationSpec:
    name: str
    requires_capability: str | None = None
    mutating: bool = True

class OperationRegistry:
    def __init__(self, specs: dict[str, OperationSpec] | None = None):
        self._specs: dict[str, OperationSpec] = dict(specs or {})

    def register(self, spec: OperationSpec) -> None:
        self._specs[spec.name] = spec

    def has(self, name: str) -> bool:
        return name in self._specs

    def spec(self, name: str) -> OperationSpec:
        if name not in self._specs:
            raise KeyError(f"unbekannte Operation: {name}")
        return self._specs[name]

def default_registry() -> OperationRegistry:
    r = OperationRegistry()
    for n, cap in [("memory.alloc", "memory.alloc"), ("memory.free", "memory.free"),
                   ("gpuvm.map", "gpuvm.map"), ("gpuvm.unmap", "gpuvm.unmap"),
                   ("gpuvm.lookup", "gpuvm.lookup"),
                   ("keymgr.claim", "keymgr.claim"), ("keymgr.release", "keymgr.release"),
                   ("keymgr.snapshot", "keymgr.snapshot"),
                   ("ccp.submit", "ccp.submit"),
                   ("kernel.snapshot", "kernel.read"),
                   ("kernel.read", "kernel.read"),
                   ("kernel.dump.full", "kernel.dump.full")]:
        r.register(OperationSpec(n, cap))
    for n in ("session.ping", "session.status"):
        r.register(OperationSpec(n, None, mutating=False))
    r.register(OperationSpec("experiment.load", None))
    for n in ("mock.hang", "mock.leak", "mock.free_fail"):
        r.register(OperationSpec(n, "memory.alloc"))
    for n in ("verify.userland.exec", "verify.kernel.read", "verify.kernel.exec",
              "verify.kernel.copyout", "verify.samu.access"):
        r.register(OperationSpec(n, None, mutating=False))
    return r

class ReliabilityStats(StrictModel):
    attempts: int = Field(ge=0)
    successes: int = Field(ge=0); clean_failures: int = Field(ge=0)
    hangs: int = Field(ge=0); panics: int = Field(ge=0); reboots: int = Field(ge=0)

            raise ValueError("successes+clean_failures+hangs+panics+reboots != attempts")

class CapabilityGrant(StrictModel):
    capability: CapabilityName
    verify_operation: Optional[VerifyOperation] = None   # C4: Pattern enforced

    def _privileged_needs_verify(self):
        if self.capability.split(".")[0] in PRIVILEGED_NAMESPACES \
                and self.verify_operation is None:
            raise ValueError(f"'{self.capability}' erfordert verify_operation")
        return self

def validate_exploit_registry_bindings(manifest, registry: OperationRegistry) -> None:
    """C4: verify_operation muss in der Registry existieren (Intake-Pflicht)."""
    errors = [g.verify_operation for g in manifest.provides
              if g.verify_operation and not registry.has(g.verify_operation)]
    if errors:
        raise ValueError(f"verify_operation nicht in Registry: {errors}")

def capability_grants_from_runs(manifest, trigger_run_id: str,
                                verification_run_id: str,
                                observed_seq: int, observed_at, context):
    """C4: PROVEN-Grants nur aus getrenntem Verifikationslauf.
    Privilegierte Namespaces: verification_run != trigger_run erzwingen."""
    if trigger_run_id == verification_run_id:
        privileged = [g for g in manifest.provides
                      if g.capability.split(".")[0] in PRIVILEGED_NAMESPACES]
        if privileged:
            raise ValueError("Verifikationslauf darf nicht der Trigger-Lauf sein "
                             f"({[g.capability for g in privileged]})")
    from .capability import CapabilityRecord, VerifiedBy
    from .common import Assurance, Availability, EvidenceRef
    out = []
    for g in manifest.provides:
        proven = g.verify_operation is not None and g.capability.split(".")[0] \

            name=g.capability,

                                   expected="nonce", observed="nonce",
                                   verification_run=verification_run_id)
                     if proven else None,
            observed_seq=observed_seq, observed_at=observed_at))
    return out

from __future__ import annotations
from datetime import datetime
from enum import Enum
from typing import Optional
from pydantic import Field, model_validator
from .common import CandidateId, EvidenceRef, Sha256, StrictModel

class CandidateState(str, Enum):
    OBSERVATION = "OBSERVATION"; ANOMALY = "ANOMALY"; BUG = "BUG"
    LOGIC_BUG = "LOGIC_BUG"; CRASH = "CRASH"; PRIMITIVE = "PRIMITIVE"
    EXPLOIT_CANDIDATE = "EXPLOIT_CANDIDATE"; EXPLOITABLE = "EXPLOITABLE"
    CHAINABLE = "CHAINABLE"; CLOSED = "CLOSED"; PARKED = "PARKED"

class StateTransition(StrictModel):
    state: CandidateState
    at: datetime
    actor: str
    evidence: Optional[EvidenceRef] = None
    note: Optional[str] = None

class ChainLink(StrictModel):
    chain_id: str
    position: int = Field(ge=0)
    status: str
    evidence: Optional[EvidenceRef] = None

class Reproducer(StrictModel):
    id: str
    expected_observation: str
    input_sha256: Optional[Sha256] = None

class CandidateRecord(StrictModel):
    id: CandidateId
    surface_id: str
    source: str
    state: CandidateState
    history: list[StateTransition] = Field(min_length=1)
    reproducers: list[Reproducer] = Field(default_factory=list)
    chain_links: list[ChainLink] = Field(default_factory=list)
    crash_fingerprints: list[Sha256] = Field(default_factory=list)
    exploit_workspace: Optional[str] = None

    @model_validator(mode="after")
    def _invariants(self):
        # C4: aktueller State == letzter History-State
        if self.history[-1].state is not self.state:
            raise ValueError("state != history[-1].state")
        # C4: history.at monoton nicht fallend
        ts = [h.at for h in self.history]
        if any(b < a for a, b in zip(ts, ts[1:])):
            raise ValueError("history.at fallend")
        # C4: Eindeutigkeiten
        if len({l.position for l in self.chain_links}) != len(self.chain_links):
            raise ValueError("chain_links.position doppelt")
        if len({r.id for r in self.reproducers}) != len(self.reproducers):
            raise ValueError("reproducer.id doppelt")
        if len(set(self.crash_fingerprints)) != len(self.crash_fingerprints):
            raise ValueError("crash_fingerprints doppelt")
        # C4: State-abhängige Mindestausstattung
        if self.state is CandidateState.CHAINABLE and not self.chain_links:
            raise ValueError("CHAINABLE erfordert mindestens einen chain_link")
        if self.state in (CandidateState.EXPLOIT_CANDIDATE,
                          CandidateState.EXPLOITABLE,
                          CandidateState.CHAINABLE) and not self.reproducers:
            raise ValueError(f"{self.state.value} erfordert reproducer")
        return self

SECRET_KEY_RE = re.compile(r"(?i)(password|pass|token|secret|api[_-]?key|key)$")

def redact_options(options: dict,
                   sensitive_from_module: list[str] | None = None,
                   extra_fields: list[str] | None = None) -> tuple[dict, list[str]]:
    """C6-Priorität:
    1. Modul-Metadaten (sensitive_options aus dem PS4RF-Modulmanifest bzw.
       module_info-Metadaten des Adapters) — Vorrang,
    2. extra_fields — expliziter Override,
    3. Regex — nur Fallback.
    Kein secret-abgeleiteter Hash wird erzeugt oder gespeichert."""
    sensitive = set(sensitive_from_module or ()) | set(extra_fields or ())
    out, redacted = {}, []
    for k, v in options.items():
        if k in sensitive or SECRET_KEY_RE.search(k):
            out[k] = REDACTED
            redacted.append(k)
        else:
            out[k] = v
    return out, sorted(redacted)
# options_sha256 unverändert: Hash nur über das bereits bereinigte Objekt.
