class CapabilityIntake:
    """V2/V13: session_id=None = Boot-Intake (nur BOOT-Records)."""
    def __init__(self, session_id: str | None, boot_id: str) -> None:
        if not boot_id:
            raise ValueError("CapabilityIntake benötigt boot_id")
        self.session_id = session_id
        self.boot_id = boot_id
        self._next_seq = 0

    def ingest(self, rec: CapabilityRecord) -> CapabilityRecord:
        if rec.observed_seq is not None:
            raise ValueError("Record trägt bereits observed_seq (Host-only Feld)")
        if rec.scope is Scope.SESSION and self.session_id is None:
            raise ValueError("Boot-Intake kann keine SESSION-Records ingestieren")
        self._next_seq += 1
        updated = rec.model_copy(update={
            "observed_seq": self._next_seq,
            "observed_at": datetime.now(timezone.utc),
            "session_id": self.session_id if rec.scope is Scope.SESSION else None,
            "boot_id": self.boot_id})
        # V2/Integrität: model_copy umgeht Validatoren — deshalb Revalidation:
        return type(rec).model_validate(updated.model_dump())

def expire_after_reboot(records, new_boot_id: str | None = None):
    """V2: KEIN Capability Record überlebt einen Reboot. BOOT gilt
    ausschließlich für genau boot_id. Persistente Fakten (Firmware/Profile)
    leben im Profile/Evidence Store und werden pro Boot NEU ingestiert."""
    removed = [r for r in records
               if r.scope is Scope.SESSION
               or new_boot_id is None or r.boot_id != new_boot_id]
    kept = [r for r in records if r not in removed]
    return kept, removed

HELLO_REQUIRED = ("label", "msg", "agent_version", "transport", "context",
                  "provider", "nonce", "boot_id")          # V3: boot_id Pflicht

class HandshakeRole(str, Enum):
    """V3: Semantische Rollen — NIEMALS aus TCP connect()/accept() ableiten."""
    AGENT = "AGENT"      # sendet HELLO, empfängt HELLO_ACK, sendet AUTH
    HOST = "HOST"        # empfängt HELLO, sendet HELLO_ACK, sendet AUTH_ACK

# Transcript unverändert: SHA256(JCS(AGENT_HELLO) || JCS(HOST_HELLO_ACK))
def transcript_hash(agent_hello: dict, host_hello_ack: dict) -> bytes:
    if agent_hello.get("label") != HANDSHAKE_LABEL:
        raise HandshakeError("HELLO.label fehlt oder falsch")
    missing = [k for k in HELLO_REQUIRED if k not in agent_hello]
    if missing:
        raise HandshakeError(f"HELLO unvollständig: {missing}")
    return hashlib.sha256(canonical_bytes(agent_hello)
                          + canonical_bytes(host_hello_ack)).digest()

class Duplex(Protocol):
    """Transportabstraktion: Rollencode kennt weder Client noch Server."""
    def send_all(self, data: bytes) -> None: ...
    def recv_exact(self, n: int) -> bytes: ...

def _send_msg(ep, obj): 
    d = canonical_bytes(obj); ep.send_all(len(d).to_bytes(4, "big") + d)

def _recv_msg(ep): 
    n = int.from_bytes(ep.recv_exact(4), "big"); 
    return json.loads(ep.recv_exact(n))

def agent_handshake(ep: Duplex, secret: bytes, hello: dict) -> tuple[HandshakeKeys, dict]:
    """AGENT-Rolle. Identisch für Reverse-TCP (Agent=TCP-Client) und
    Listener (Agent=TCP-Server)."""
    if len(secret) != SECRET_LEN: raise ProtocolError("secret 32 Byte")
    _send_msg(ep, hello)
    ack = _recv_msg(ep)
    th = transcript_hash(hello, ack)
    keys = derive_keys(secret, bytes.fromhex(hello["nonce"]),
                       bytes.fromhex(ack["nonce"]), th)
    _send_msg(ep, {"msg": "AUTH",
                   "mac": proof(keys.k_hs, "agent", th).hex()})
    auth_ack = _recv_msg(ep)
    if auth_ack.get("mac") != proof(keys.k_hs, "host", th).hex():
        raise HandshakeError("host proof invalid")
    return keys, ack

def host_handshake(ep: Duplex, secret: bytes, accept: dict) -> tuple[HandshakeKeys, dict]:
    """HOST-Rolle. accept = HELLO_ACK-Inhalt (limits, aead, session_id)."""
    hello = _recv_msg(ep)
    _send_msg(ep, accept)
    th = transcript_hash(hello, accept)
    keys = derive_keys(secret, bytes.fromhex(hello["nonce"]),
                       bytes.fromhex(accept["nonce"]), th)
    auth = _recv_msg(ep)
    if auth.get("mac") != proof(keys.k_hs, "agent", th).hex():
        raise HandshakeError("agent proof invalid")
    _send_msg(ep, {"msg": "AUTH_ACK",
                   "mac": proof(keys.k_hs, "host", th).hex()})
    return keys, hello

        buf = b""

"""V8: Manuell freigebbares Gate für deterministische Check/Mutate/Use-Fenster.
V10: Destroy liefert Waitern RESOURCE_DESTROYED, niemals normalen Success."""
from __future__ import annotations
import threading, time
from enum import Enum

class GateWaitResult(str, Enum):
    RELEASED = "RELEASED"; DESTROYED = "DESTROYED"; TIMEOUT = "TIMEOUT"

class GateDestroyed(Exception): ...

class Gate:
    def __init__(self, expected: int) -> None:
        self._cond = threading.Condition()
        self._arrived = 0
        self._expected = expected
        self._released = False
        self._destroyed = False

    def arrive(self) -> int:                             # V8: atomar
        with self._cond:
            if self._destroyed:
                raise GateDestroyed("gate destroyed")
            self._arrived += 1
            self._cond.notify_all()
            return self._arrived

    def wait(self, timeout_ms: int) -> GateWaitResult:
        """V8: blockiert bis RELEASE — NICHT bis parties erreicht."""
        deadline = time.monotonic() + timeout_ms / 1000
        with self._cond:
            while not (self._released or self._destroyed):
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    return GateWaitResult.TIMEOUT        # V10: kein Zombie-Thread
                self._cond.wait(remaining)
            return (GateWaitResult.DESTROYED if self._destroyed
                    else GateWaitResult.RELEASED)        # V10: DESTROYED ≠ Success

    def status(self) -> dict:
        """V8: arrived/expected OHNE Freigabe."""
        with self._cond:
            return {"arrived": self._arrived, "expected": self._expected,
                    "released": self._released, "destroyed": self._destroyed}

    def release(self) -> None:
        with self._cond:
            self._released = True
            self._cond.notify_all()

    def destroy(self) -> None:                           # V10: wake all
        with self._cond:
            self._destroyed = True
            self._cond.notify_all()

import queue, threading

class BoundedPool:
    def __init__(self, size: int, max_inflight: int) -> None:
        self._queue: queue.Queue = queue.Queue(maxsize=max_inflight)
        self._lock = threading.Lock()
        self._poisoned = False
        self._in_flight: dict[str, str] = {}      # rid → op (IN_FLIGHT_AT_RECOVERY)
        self._pending: list[tuple] = []
        for _ in range(size):
            threading.Thread(target=self._worker, daemon=True).start()

    @property
    def poisoned(self) -> bool:
        with self._lock: return self._poisoned

    def submit(self, rid: str, op: str, fn) -> str:
        """'OK' | 'TOO_MANY_INFLIGHT' | 'POISONED'"""
        with self._lock:
            if self._poisoned:
                return "POISONED"
            try:
                self._queue.put_nowait((rid, op, fn))
            except queue.Full:
                return "TOO_MANY_INFLIGHT"
            self._pending.append((rid, op))
            return "OK"

    def poison(self, reason: str) -> list[dict]:
        """V5: atomar. Drain gequeueter mutierender Requests; in-flight
        bleiben und werden als IN_FLIGHT_AT_RECOVERY protokolliert."""
        with self._lock:
            self._poisoned = True
            in_flight = [{"request_id": rid, "operation": op,
                          "status": "IN_FLIGHT_AT_RECOVERY"}
                         for rid, op in self._in_flight.items()]
            self._pending.clear()                 # gequeued, nie gestartet → rejected
            return in_flight

    def _worker(self):
        while True:
            rid, op, fn = self._queue.get()
            with self._lock:
                if self._poisoned:                # V5: Check UNMITTELBAR vor Entry
                    self._queue.task_done(); continue
                self._in_flight[rid] = op
            try:
                fn()
            finally:
                with self._lock:
                    self._in_flight.pop(rid, None)
                self._queue.task_done()

"""V7: Eindeutige Request-IDs pro Session; bounded completed cache."""
import collections, re, threading
from .handshake import ProtocolError

REQ_ID_RE = re.compile(r"^REQ-[0-9a-f]{32}$")

class RequestIdRegistry:
    def __init__(self, completed_cache_size: int = 4096) -> None:
        self._lock = threading.Lock()
        self._pending: set[str] = set()
        self._completed: collections.OrderedDict[str, None] = collections.OrderedDict()
        self._cache_size = completed_cache_size

    def register(self, rid: str) -> None:
        with self._lock:
            if not REQ_ID_RE.match(rid):
                raise ProtocolError(f"request_id Format: {rid!r}")
            if rid in self._pending or rid in self._completed:
                raise ProtocolError("DUPLICATE_REQUEST_ID")   # V7: dup → Session close
            self._pending.add(rid)

    def complete(self, rid: str) -> None:
        with self._lock:
            self._pending.discard(rid)
            self._completed[rid] = None
            while len(self._completed) > self._cache_size:
                self._completed.popitem(last=False)

    def is_pending(self, rid: str) -> bool:
        with self._lock: return rid in self._pending

class AgentRpcBudget:
    """V9: Agent-seitig. Enforce: nach Decrypt vor Dispatch (charge_in),
    vor Encrypt der Response/Event (charge_out). Session- und Experiment-Level."""
    def __init__(self, session_max: int, max_frame_bytes: int,
                 experiment_max_rpc_bytes: int | None = None):
        self.max_rpc_bytes = session_max
        self.max_frame_bytes = max_frame_bytes
        self.experiment_max = experiment_max_rpc_bytes
        self._used = 0
        self._exp_used = 0
        self._lock = threading.Lock()

    def _charge(self, n: int, exp: bool) -> None:
        with self._lock:
            self._used += n
            if self._used > self.max_rpc_bytes:
                raise BudgetExceeded("session rpc budget exceeded")
            if exp:
                self._exp_used += n
                if self._experiment_max and self._exp_used > self._experiment_max:
                    raise BudgetExceeded("experiment rpc budget exceeded")

    def charge_in(self, plaintext: bytes, *, experiment: bool = False):
        if len(plaintext) > self.max_frame_bytes:
            raise BudgetExceeded("frame over max_frame_bytes")
        self._charge(len(plaintext), experiment)

    def charge_out(self, plaintext: bytes, *, experiment: bool = False):
        self._charge(len(plaintext), experiment)
