def resolve(name: str, scope: Scope, context: TargetExecutionContext,
            records: Sequence[CapabilityRecord], authority: AuthorityTable,
            manual: Optional[ManualOverride] = None,
            *, session_id: Optional[str] = None,
            boot_id: Optional[str] = None) -> Resolution:
    """R1: Nur Records mit EXAKT name/scope/context (+ session_id bei SESSION,
    + boot_id) nehmen teil. Gemischte Records werden ausgeschlossen und
    auditiert (fail-closed wäre alternative Policy; Filter+Audit ist hier
    gewählt, damit Negotiation pro Context sauber iterieren kann)."""
    if scope is Scope.SESSION and not session_id:
        raise ValueError("resolve(SESSION) benötigt session_id")
    if boot_id is None:
        raise ValueError("resolve benötigt boot_id")

    participants: list[tuple[int, CapabilityRecord, ProviderRule]] = []
    excluded: list[ExcludedRecord] = []
    for i, rec in enumerate(records):
        if rec.name != name:
            excluded.append(ExcludedRecord(i, "NAME_MISMATCH")); continue
        if rec.scope is not scope:
            excluded.append(ExcludedRecord(i, "SCOPE_MISMATCH")); continue
        if rec.context is not context:
            excluded.append(ExcludedRecord(i, "CONTEXT_MISMATCH")); continue
        if scope is Scope.SESSION and rec.session_id != session_id:
            excluded.append(ExcludedRecord(i, "SESSION_MISMATCH")); continue
        if rec.boot_id != boot_id:
            excluded.append(ExcludedRecord(i, "BOOT_MISMATCH")); continue
        if rec.observed_seq is None:
            raise ValueError(f"Record {i} nicht durch CapabilityIntake ingestiert")
        rule = authority.rule_for(name, rec.provider_type, rec.provider)
        if rule is not None:
            participants.append((i, rec, rule))
        # Records ohne Authority-Regel: weiter ignoriert (AUTHORITY_INVALID, geloggt)

    seqs = [r.observed_seq for _, r, _ in participants]
    if len(set(seqs)) != len(seqs):
        raise ValueError("Integrity: doppelte observed_seq unter Teilnehmern "
                         "(Intake-Verstoß)")

    # --- Availability (v0.5-Logik, jetzt über gefilterte Teilnehmer) ---
    if manual is not None and manual.availability is not None:
        availability = manual.availability
        a_res = AxisResolution(availability.value, AxisFactor.MANUAL_OVERRIDE,
                               detail=manual.detail)
    else:
        elig = [(i, r, ru) for i, r, ru in participants if ru.availability_write]
        if not elig:
            availability = Availability.UNKNOWN
            a_res = AxisResolution("UNKNOWN", AxisFactor.NO_RECORD)
        else:
            top = max(ru.availability_rank for _, _, ru in elig)
            tier = sorted([(i, r, ru) for i, r, ru in elig
                           if ru.availability_rank == top],
                          key=lambda t: t[1].observed_seq)   # seq, nie provider-Zeit
            i, r, _ = tier[-1]
            factor = (AxisFactor.PROVIDER_AUTHORITY if len(tier) == 1
                      else AxisFactor.RUNTIME_RECENCY)
            availability = r.availability
            a_res = AxisResolution(availability.value, factor, i)

    # --- Assurance (v0.5-Logik unverändert, Teilnehmer gefiltert) ---
    best_rank, best = -1, None
    for i, r, ru in participants:
        eff = _clamp(r.assurance, ru.max_assurance)
        if ASSURANCE_RANK[eff] > best_rank:
            best_rank, best = ASSURANCE_RANK[eff], (i, r, eff)
    manual_assur = None
    if manual is not None and manual.assurance is not None:
        if manual.assurance is Assurance.PROVEN and (
                manual.evidence is None or manual.verified_by is None):
            raise ValueError("MANUAL darf PROVEN nur mit evidence + verified_by setzen")
        op_rule = authority.rule_for(name, ProviderType.OPERATOR, "manual")
        manual_assur = _clamp(manual.assurance,
                              op_rule.max_assurance if op_rule else Assurance.OBSERVED)
    if manual_assur is not None and ASSURANCE_RANK[manual_assur] >= best_rank:
        assurance = manual_assur
        s_res = AxisResolution(assurance.value, AxisFactor.MANUAL_OVERRIDE)
    elif best is None:
        assurance = Assurance.UNVERIFIED
        s_res = AxisResolution("UNVERIFIED", AxisFactor.NO_RECORD)
    else:
        same = sorted([(i, r, _clamp(r.assurance, ru.max_assurance))
                       for i, r, ru in participants
                       if ASSURANCE_RANK[_clamp(r.assurance, ru.max_assurance)] == best_rank],
                      key=lambda t: t[1].observed_seq)
        i, r, eff = same[-1]
        factor = (AxisFactor.RECENCY if len(same) > 1 else
                  AxisFactor.AUTHORITY_CEILING if eff is not r.assurance
                  else AxisFactor.EVIDENCE_RANK)
        assurance = eff
        s_res = AxisResolution(assurance.value, factor, i)

    return Resolution(name, availability, assurance, a_res, s_res,
                      tuple(excluded))

def expire_after_reboot(records: Sequence[CapabilityRecord]
                        ) -> tuple[list[CapabilityRecord], list[CapabilityRecord]]:
    """R1: Nach Reboot verfallen alle SESSION-Records sowie BOOT-Records mit
    expires_on_reboot=True. Rückgabe: (verbleibend, entfernt)."""
    kept = [r for r in records
            if r.scope is Scope.BOOT and not r.expires_on_reboot]
    removed = [r for r in records if r not in kept]
    return kept, removed

from dataclasses import dataclass
from enum import Enum

from pydantic import model_validator
from .capability import CapabilityIntake, CapabilityRecord, Scope, VerifiedBy
from .common import (CapabilityName, OperationName, ProviderType, RunId, Sha256,
                     StrictModel, TargetExecutionContext)

class VerificationOutcome(str, Enum):
    PASS = "PASS"; FAIL = "FAIL"; ERROR = "ERROR"; TIMEOUT = "TIMEOUT"

class VerificationEvidence(StrictModel):
    verification_run_id: RunId
    operation: OperationName          # enforce verify.* unten
    outcome: VerificationOutcome
    capability: CapabilityName
    provider: str                     # Exploit-Identität
    provider_type: ProviderType = ProviderType.EXPLOIT
    profile_sha256: Sha256            # Target/Profil-Binding
    context: TargetExecutionContext
    scope: Scope
    session_id: Optional[str] = None
    boot_id: str
    expected: str
    observed: str
    evidence_sha256: Sha256           # Hash im Evidence Store

    @model_validator(mode="after")
    def _verify_namespace(self):
        if not self.operation.startswith("verify."):
            raise ValueError("verification.operation muss verify.* sein")
        if self.scope is Scope.SESSION and not self.session_id:
            raise ValueError("SESSION-Verifikation benötigt session_id")
        return self

class EvidenceStore(Protocol):
    def hash_is_valid(self, sha256: str) -> bool: ...

@dataclass(frozen=True)
class VerificationRejection(Exception):
    reason: str    # REGISTRY_UNKNOWN | CAPABILITY_MISMATCH | OPERATION_MISMATCH |
                   # OUTCOME_NOT_PASS | RUN_NOT_SEPARATE | PROFILE_MISMATCH |
                   # EVIDENCE_HASH_INVALID
    detail: str = ""

def grant_from_verification(verification: VerificationEvidence,
                            capability: CapabilityName,
                            manifest_provides: list,       # list[CapabilityGrant]
                            trigger_run_id: RunId,
                            expected_profile_sha256: Sha256,
                            registry,                       # OperationRegistry
                            evidence_store: EvidenceStore,
                            intake: CapabilityIntake) -> CapabilityRecord:
    """R2: PROVEN nur wenn ALLE Prüfungen bestehen. PRIVILEGIERT zusätzlich:
    verification_run_id != trigger_run_id."""
    from .common import PRIVILEGED_NAMESPACES, Assurance, Availability

    grant = next((g for g in manifest_provides if g.capability == capability), None)
    if grant is None:
        raise VerificationRejection("CAPABILITY_MISMATCH",
                                    f"{capability} nicht in manifest.provides")
    if not registry.has(verification.operation) or \
            registry.spec(verification.operation).name.split(".")[0] != "verify":
        raise VerificationRejection("REGISTRY_UNKNOWN", verification.operation)
    if verification.capability != capability:
        raise VerificationRejection("CAPABILITY_MISMATCH",
                                    f"{verification.capability} != {capability}")
    if grant.verify_operation is None or \
            verification.operation != grant.verify_operation:
        raise VerificationRejection("OPERATION_MISMATCH",
                                    f"{verification.operation} != {grant.verify_operation}")
    if verification.outcome is not VerificationOutcome.PASS:
        raise VerificationRejection("OUTCOME_NOT_PASS", verification.outcome.value)
    privileged = capability.split(".")[0] in PRIVILEGED_NAMESPACES
    if privileged and verification.verification_run_id == trigger_run_id:
        raise VerificationRejection("RUN_NOT_SEPARATE",
                                    "Verifikationslauf == Trigger-Lauf (privilegiert)")
    if verification.profile_sha256 != expected_profile_sha256:
        raise VerificationRejection("PROFILE_MISMATCH", "Target/Profil-Binding")
    if not evidence_store.hash_is_valid(verification.evidence_sha256):
        raise VerificationRejection("EVIDENCE_HASH_INVALID", verification.evidence_sha256)

    proven = privileged            # nicht-privilegierte: OBSERVED (kein verified_by nötig)
    return intake.ingest(CapabilityRecord(
        name=capability,
        availability=Availability.AVAILABLE,
        assurance=Assurance.PROVEN if proven else Assurance.OBSERVED,
        context=verification.context,
        provider=verification.provider,
        provider_type=verification.provider_type,
        scope=verification.scope,
        session_id=verification.session_id,
        boot_id=verification.boot_id,
        evidence=None,             # wird vom Store-Intake mit EvidenceRef belegt
        verified_by=VerifiedBy(operation=verification.operation,
                               expected=verification.expected,
                               observed=verification.observed,
                               verification_run=verification.verification_run_id)
                    if proven else None))

def negotiate(self, hello_context: TargetExecutionContext) -> tuple[list[dict], list[dict]]:
    """R3: Exakt effective_grant() je Capability. Kein availability-Shortcut,
    kein HOMEBREW-Hardcode — Context kommt aus dem authentisierten HELLO."""
    auth = default_authority()
    granted, denied = [], []
    ns_of = lambda name: name.split(".")[0]
    for name in sorted({r.name for r in self.capabilities}):
        recs = [r for r in self.capabilities if r.name == name]
        if not recs:
            continue
        res = resolve(name, Scope.SESSION, hello_context, recs, auth,
                      session_id=self._session_id, boot_id=self._boot_id)
        req = Requirement(name, auth.min_assurance_for_grant.get(
            ns_of(name), Assurance.UNVERIFIED))
        d = effective_grant(req, res, hello_context, hello_context, auth,
                            recovery_required=self.console_recovery)
        entry = {"name": name, "availability": res.availability.value,
                 "assurance": res.assurance.value}
        (granted if d.granted else denied).append(
            entry if d.granted else {"name": name, "reason": d.reason.value,
                                     **entry})
    return granted, denied

"""R6: Nur nachweislich FREED Ressourcen verlassen den Manager."""
from __future__ import annotations
import os
from dataclasses import dataclass, field
from enum import Enum

class CleanupResult(str, Enum):
    FREED = "FREED"; FAILED = "CLEANUP_FAILED"; UNCERTAIN = "UNCERTAIN"

@dataclass
class ManagedResource:
    ref: str
    kind: str
    recoverable: bool = True
    force_cleanup_failure: bool = False
    cleanup_attempted: bool = False
    cleanup_result: CleanupResult | None = None

@dataclass
class CleanupReport:
    state: str                                  # COMPLETE | PARTIAL | FAILED
    resources: list[dict] = field(default_factory=list)
    console_dirty: bool = False

class ResourceManager:
    def __init__(self) -> None:
        self._held: dict[str, ManagedResource] = {}
        self._leaked: dict[str, ManagedResource] = {}   # R6: Evidence bleibt erhalten

    def acquire(self, kind: str, *, recoverable: bool = True,
                force_cleanup_failure: bool = False) -> str:
        ref = f"{kind}:{os.urandom(8).hex().upper()}"
        self._held[ref] = ManagedResource(ref, kind, recoverable,
                                          force_cleanup_failure)
        return ref

    def release(self, ref: str) -> bool:
        r = self._held.get(ref)
        if r is None:
            return False
        r.cleanup_attempted = True
        if r.force_cleanup_failure or not r.recoverable:
            r.cleanup_result = (CleanupResult.FAILED if r.recoverable
                                else CleanupResult.UNCERTAIN)
            self._leaked[ref] = self._held.pop(ref)
            return False
        r.cleanup_result = CleanupResult.FREED
        del self._held[ref]                          # nur FREED verschwindet
        return True

    def cleanup_all(self) -> CleanupReport:
        """R6: Cleanup-Versuch für alle gehaltenen Ressourcen.
        FAILED/UNCERTAIN bleiben als Evidence erhalten (self._leaked);
        jede nicht-FREED Ressource macht die Konsole dirty."""
        for ref in list(self._held):
            self.release(ref)

def _barrier_create(self, args):
    bid = "barrier:" + os.urandom(8).hex().upper()
    self._barriers[bid] = {"parties": int(args["parties"]),
                           "event": threading.Event(),
                           "arrived": 0, "released": False}
    return {"barrier": bid}

def _barrier_wait(self, args):
    b = self._barriers.get(args["barrier"])
    if b is None:
        return err("RESOURCE_STATE_INVALID")
    b["arrived"] += 1
    if b["arrived"] >= b["parties"]:
        b["released"] = True
        b["event"].set()
        return {"arrived": True}
    if not b["event"].wait(timeout_ms=args.get("timeout_ms", 5000) / 1000):
        return err("BARRIER_TIMEOUT")
    return {"arrived": True, "released": b["released"]}

def _barrier_release(self, args):
    b = self._barriers.get(args["barrier"])
    if b is None:
        return err("RESOURCE_STATE_INVALID")
    b["released"] = True
    b["event"].set()
    return {}

def _barrier_destroy(self, args):
    b = self._barriers.pop(args["barrier"], None)
    if b is None:
        return err("RESOURCE_STATE_INVALID")
    b["event"].set()                     # wartende werden entriegelt
    return {}

def recv_frame(r, ch, max_frame_bytes: int):
    """R8: Längenprüfung VOR jeder Allokation. Unbekannter msg_type und
    jede Frame-Verletzung → ProtocolError → Session sofort schließen."""
    hdr = _recv_exact(r, 13)                     # 4 len || 1 type || 8 seq
    n = int.from_bytes(hdr[0:4], "big")
    t_byte = hdr[4]
    seq = int.from_bytes(hdr[5:13], "big")
    if t_byte not in FRAME_TYPE_NAMES:
        raise ProtocolError(f"unbekannter msg_type {t_byte}")
    if n > max_frame_bytes:
        raise ProtocolError(f"frame {n} > max_frame_bytes {max_frame_bytes}")
    mtype = FRAME_TYPE_NAMES[t_byte]
    return mtype, seq, ch.open(mtype, seq, _recv_exact(r, n))

def check_budget(ir: dict, region_sizes: dict | None = None) -> dict:
    computed = {
        "max_expanded_steps": expanded_steps(ir["steps"]),
        "max_parallel_tasks": peak_tasks(ir["steps"]),
        "max_nesting_depth": nesting_depth(ir["steps"]),
        "max_runtime_budget_ms": runtime_bound_ms(
            ir["steps"], ir["timeouts"]["default_step_ms"]),
        "max_snapshot_bytes": snapshot_bytes(ir["steps"], region_sizes),
    }
    declared = ir.get("budget", {})
    timeouts = ir["timeouts"]
    # R9: computed runtime bound muss AUCH <= timeouts.global_ms sein
    if computed["max_runtime_budget_ms"] > timeouts["global_ms"]:
        raise BudgetExceeded(
            f"runtime_bound {computed['max_runtime_budget_ms']} > "
            f"timeouts.global_ms {timeouts['global_ms']}")
    # max_rpc_bytes: declared hard cap (kein Operation-Size-Modell) —
    # Laufzeit-Enforcement kumulativ über RpcByteCounter (unten)
    if "max_rpc_bytes" in declared and declared["max_rpc_bytes"] > CAPS["max_rpc_bytes"]:
        raise BudgetExceeded("max_rpc_bytes über globalem Cap")
    for k, cap in CAPS.items():
        if k == "max_rpc_bytes":
            continue
        v = computed[k]
        if v > cap or (k in declared and v > declared[k]):
            raise BudgetExceeded(f"{k}: computed={v}, cap={cap}, "
                                 f"declared={declared.get(k)}")
    return computed

"""R9: Kumulatives RPC-Budget über canonical PLAINTEXT bytes — Requests UND
Responses/Events, für die gesamte Experiment-Session. max_frame_bytes ist
ein separates Transportlimit pro Frame."""
class RpcByteBudgetExceeded(BudgetExceeded): ...

class RpcByteCounter:
    def __init__(self, max_rpc_bytes: int, max_frame_bytes: int) -> None:
        self.max_rpc_bytes = max_rpc_bytes
        self.max_frame_bytes = max_frame_bytes
        self.used = 0

    def charge(self, plaintext: bytes, *, direction: str) -> int:
        if len(plaintext) > self.max_frame_bytes:
            raise RpcByteBudgetExceeded(
                f"frame {len(plaintext)} > max_frame_bytes {self.max_frame_bytes}")
        self.used += len(plaintext)
        if self.used > self.max_rpc_bytes:
            raise RpcByteBudgetExceeded(
                f"rpc bytes {self.used} > max_rpc_bytes {self.max_rpc_bytes} "
                f"(nach {direction})")
        return self.used
# Engine: charge() vor seal() (Request) und nach open() (Response/Event).

import re as _re
_HEX16 = _re.compile(r"^0x[0-9a-f]{16}$")

def normalize_u64(value: str | int) -> str:
    """R11: Kanonische Darstellung '0x' + exakt 16 lowercase Hexstellen.
    Für Ingest-Pfade mit variabler Breite (Tooling-Importe); Schemas erzwingen
    die kanonische Form bereits, variable Breite wird dort abgelehnt."""
    v = int(value, 16) if isinstance(value, str) else int(value)
    if isinstance(value, str) and not value.lower().startswith("0x"):
        raise ValueError("u64 erwartet hexadezimal mit 0x-Präfix")
    if not (0 <= v < 2**64):
        raise ValueError(f"kein u64: {value!r}")
    return f"0x{v:016x}"

def is_canonical_u64(value: str) -> bool:
    return bool(_HEX16.match(value))
